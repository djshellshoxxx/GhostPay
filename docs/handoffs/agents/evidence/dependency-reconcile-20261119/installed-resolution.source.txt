import { createRequire } from 'node:module';
import { readFileSync } from 'node:fs';
import { resolve } from 'node:path';
import { pathToFileURL } from 'node:url';
import assert from 'node:assert/strict';
const consumer = createRequire(resolve('packages/confidential-amounts/package.json'));
const zama = createRequire(consumer.resolve('@zama-fhe/sdk'));
const poolConsumer = createRequire(resolve('packages/external-pools-privacypools/package.json'));
const privacyPools = createRequire(poolConsumer.resolve('@0xbow/privacy-pools-core-sdk'));
const origins = { ghostpay: consumer.resolve('viem'), zama: zama.resolve('viem'), privacyPools: privacyPools.resolve('viem') };
assert.equal(new Set(Object.values(origins)).size, 1);
const viem = JSON.parse(readFileSync('node_modules/viem/package.json', 'utf8')).version;
assert.equal(viem, '2.57.19');
const vitestVersion = JSON.parse(readFileSync('node_modules/vitest/package.json', 'utf8')).version;
const coverageVersion = JSON.parse(readFileSync('node_modules/@vitest/coverage-v8/package.json', 'utf8')).version;
assert.equal(vitestVersion, '5.1.0');
assert.equal(coverageVersion, vitestVersion);
const vitest = createRequire(resolve('node_modules/vitest/package.json'));
const coverage = vitest.resolve('@vitest/coverage-v8');
await import(pathToFileURL(coverage).href);
console.log(JSON.stringify({ viemVersion: viem, viemOrigins: origins, vitestVersion, coverageVersion, coverageResolvedFromVitest: coverage, coverageImport: 'pass' }, null, 2));
